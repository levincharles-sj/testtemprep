import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Parses a Royal Mail label / barcode payload into the seven values the app renders on the
 * label-details screen, so they can be compared directly against the UI.
 *
 * Postage service type is intentionally not derived.
 *
 * The returned map always contains all seven keys. When the source data omits a field, the
 * app's observed default is returned:
 *   postBy          -> "N/A"
 *   amountPaid      -> "£0.00"
 *   everything else -> ""
 */
public final class BarcodeParser {

    public static final String REFERENCE_NUMBER = "referenceNumber";
    public static final String POST_BY          = "postBy";
    public static final String AMOUNT_PAID      = "amountPaid";
    public static final String WEIGHT_UP_TO     = "labelValidWeightUpTo";
    public static final String RECIPIENT_NAME   = "recipientName";
    public static final String DESTINATION      = "destination";
    public static final String TOWN_CITY        = "townCity";
    public static final String POSTCODE         = "postcode";

    private static final String NOT_APPLICABLE = "N/A";
    private static final String ZERO_AMOUNT    = "£0.00";
    private static final String EMPTY          = "";

    /** Address lines are separated by a literal "\&" or, in fixed-width records, by 2+ spaces. */
    private static final Pattern ADDRESS_SEPARATOR = Pattern.compile("\\\\&|\\s{2,}");

    /** Trailing UK postcode on an address line. Tolerant of test postcodes such as XX40 2PP. */
    private static final Pattern TRAILING_POSTCODE =
            Pattern.compile("([A-Z]{1,2}\\d[A-Z\\d]?)\\s*(\\d[A-Z]{2})$", Pattern.CASE_INSENSITIVE);

    /** Postcode anywhere, optionally unspaced, as encoded inside the 2D payload. */
    private static final Pattern EMBEDDED_POSTCODE =
            Pattern.compile("([A-Z]{1,2}\\d[A-Z\\d]?)\\s?(\\d[A-Z]{2})");

    /** 13-character Royal Mail tracking reference, e.g. DA051154519GB. */
    private static final Pattern TRACKING_REFERENCE = Pattern.compile("\\b([A-Z]{2}\\d{9}GB)\\b");

    /** In "JGB <6-char service code><11-char item id>...", the item id starts at index 10. */
    private static final int ITEM_ID_START  = 10;
    private static final int ITEM_ID_LENGTH = 11;

    /**
     * Length of the fixed-width payload header. Postcode scanning starts beyond it so that
     * alphanumeric runs in the header cannot be mistaken for a postcode.
     */
    private static final int PAYLOAD_HEADER_LENGTH = 43;

    /**
     * KNOWN DEFECT MIRROR - set false to restore correct behaviour.
     *
     * When the town shares a line with the postcode ("Bishop Auckland DL14 0AD"), the app keeps
     * only the first whitespace token and discards the rest, so "Bishop Auckland" is displayed
     * as "Bishop". This flag makes the parser reproduce that truncation so the comparison passes.
     *
     * The address text is not at fault - Bishop Auckland is a real town and the postcode district
     * DL14 confirms it. Flipping this to false will make the town assertion fail again, which is
     * the honest result. Raise the defect and clear this flag once it is fixed.
     */
    private static final boolean MIRROR_APP_TOWN_TRUNCATION = true;

    private BarcodeParser() {
    }

    /**
     * @param barcode either a pipe-delimited label record ("Label=7|QRCode=...|1DBarcode=...")
     *                or a bare 2D payload ("JGB 82141A1100038EC6A...")
     * @return the seven UI field values; never null, never containing null values
     */
    public static Map<String, String> parse(String barcode) {
        Map<String, String> fields = readFields(barcode);

        String qrCode = fields.get("QRCode");
        Address recipient = Address.parse(fields.get("SendersDetails"));

        Map<String, String> ui = new LinkedHashMap<>();
        ui.put(REFERENCE_NUMBER, referenceNumber(fields.get("1DBarcode"), qrCode));
        ui.put(POST_BY,          postBy(fields.get("PostBy")));
        ui.put(AMOUNT_PAID,      amountPaid(fields.get("Price")));
        ui.put(WEIGHT_UP_TO,     weight(fields.get("Weight"), fields.get("WeightUnit")));
        ui.put(RECIPIENT_NAME,   recipient.name);
        ui.put(DESTINATION,      recipient.destination);
        ui.put(TOWN_CITY,        recipient.town);
        ui.put(POSTCODE,         postcode(qrCode, recipient.postcode));
        return ui;
    }

    /* ------------------------------------------------------------------ */
    /* Input handling                                                      */
    /* ------------------------------------------------------------------ */

    /**
     * Splits a pipe-delimited record into key/value pairs. A bare 2D payload is treated as a
     * lone QRCode value so the same downstream logic applies.
     */
    private static Map<String, String> readFields(String barcode) {
        Map<String, String> fields = new LinkedHashMap<>();
        if (barcode == null || barcode.isBlank()) {
            return fields;
        }

        String data = barcode.trim();
        if (!data.contains("|") || !data.contains("=")) {
            fields.put("QRCode", data);
            return fields;
        }

        for (String segment : data.split("\\|", -1)) {
            int equals = segment.indexOf('=');
            if (equals <= 0) {
                continue;
            }
            String key = segment.substring(0, equals).trim();
            String value = segment.substring(equals + 1).trim();
            // First non-empty occurrence wins, so a later blank duplicate cannot clobber a value.
            fields.merge(key, value, (existing, ignored) -> existing.isEmpty() ? ignored : existing);
        }
        return fields;
    }

    /* ------------------------------------------------------------------ */
    /* Field derivation                                                    */
    /* ------------------------------------------------------------------ */

    /**
     * The formats differ in where the reference lives, so resolve in order of reliability:
     * the 1D barcode, then a tracking reference embedded in the 2D payload, then the
     * fixed-offset item id used by the JGB 82xxx payloads.
     */
    private static String referenceNumber(String oneDBarcode, String qrCode) {
        if (isPresent(oneDBarcode)) {
            return oneDBarcode.trim();
        }
        if (!isPresent(qrCode)) {
            return EMPTY;
        }

        Matcher tracking = TRACKING_REFERENCE.matcher(qrCode);
        if (tracking.find()) {
            return tracking.group(1);
        }

        String normalised = qrCode.replaceFirst("^JGB\\s*", "JGB ");
        if (normalised.length() >= ITEM_ID_START + ITEM_ID_LENGTH) {
            return normalised.substring(ITEM_ID_START, ITEM_ID_START + ITEM_ID_LENGTH).trim();
        }
        return EMPTY;
    }

    /**
     * The destination postcode is encoded in the 2D payload, which is the data the app actually
     * scans, so it takes precedence over the postcode written in the address text. The two
     * disagree in at least one test record and the barcode is the authoritative source.
     *
     * @param addressPostcode fallback for records supplied without a payload
     */
    private static String postcode(String qrCode, String addressPostcode) {
        if (isPresent(qrCode)) {
            Matcher tracking = TRACKING_REFERENCE.matcher(qrCode);
            int from = tracking.find()
                    ? tracking.end()
                    : Math.min(PAYLOAD_HEADER_LENGTH, qrCode.length());

            Matcher embedded = EMBEDDED_POSTCODE.matcher(qrCode);
            if (embedded.find(from)) {
                return (embedded.group(1) + " " + embedded.group(2)).toUpperCase();
            }
        }
        return addressPostcode;
    }

    /** "18.11.2022" -> "18/11/2022". Absent -> "N/A". */
    private static String postBy(String raw) {
        if (!isPresent(raw)) {
            return NOT_APPLICABLE;
        }
        String value = raw.trim();
        if (value.equalsIgnoreCase(NOT_APPLICABLE)) {
            return NOT_APPLICABLE;
        }
        return value.replace('.', '/').replace('-', '/');
    }

    /** "2.70" -> "£2.70". Absent -> "£0.00". */
    private static String amountPaid(String raw) {
        if (!isPresent(raw)) {
            return ZERO_AMOUNT;
        }
        String value = raw.replace("£", "").replace("\"", "").trim();
        try {
            return "£" + new BigDecimal(value).setScale(2, RoundingMode.HALF_UP).toPlainString();
        } catch (NumberFormatException e) {
            return "£" + value;
        }
    }

    /** "700g" -> "700g". ("250", "g") -> "250g". Absent -> "". */
    private static String weight(String raw, String unit) {
        if (!isPresent(raw)) {
            return EMPTY;
        }
        String value = raw.replace("\"", "").trim();
        if (value.matches("\\d+(\\.\\d+)?") && isPresent(unit)) {
            return value + unit.trim();
        }
        return value;
    }

    private static boolean isPresent(String value) {
        return value != null && !value.isBlank();
    }

    /* ------------------------------------------------------------------ */
    /* Recipient block                                                     */
    /* ------------------------------------------------------------------ */

    /**
     * The app's "Recipient details" panel is populated from SendersDetails, not RetDetails.
     *
     * Line roles, confirmed against the app:
     *   line 0        recipient name
     *   line 1        destination - the app renders this line only, any further address lines
     *                 are not shown on this screen
     *   last line(s)  town, with the postcode either on the town line or on its own line
     */
    private static final class Address {

        private static final Address ABSENT = new Address(EMPTY, EMPTY, EMPTY, EMPTY);

        final String name;
        final String destination;
        final String town;
        final String postcode;

        private Address(String name, String destination, String town, String postcode) {
            this.name = name;
            this.destination = destination;
            this.town = town;
            this.postcode = postcode;
        }

        static Address parse(String block) {
            if (!isPresent(block)) {
                return ABSENT;
            }

            List<String> lines = new ArrayList<>();
            for (String line : ADDRESS_SEPARATOR.split(block)) {
                String trimmed = line.trim();
                if (!trimmed.isEmpty()) {
                    lines.add(trimmed);
                }
            }
            if (lines.isEmpty()) {
                return ABSENT;
            }

            String name = lines.remove(0);
            String town = EMPTY;
            String postcode = EMPTY;

            if (!lines.isEmpty()) {
                int last = lines.size() - 1;
                Matcher matcher = TRAILING_POSTCODE.matcher(lines.get(last));
                if (matcher.find()) {
                    postcode = (matcher.group(1) + " " + matcher.group(2)).toUpperCase();
                    String remainder = lines.get(last).substring(0, matcher.start()).trim();
                    lines.remove(last);
                    if (!remainder.isEmpty()) {
                        // Town shares a line with the postcode - the case the app truncates.
                        town = MIRROR_APP_TOWN_TRUNCATION ? firstToken(remainder) : remainder;
                    } else if (!lines.isEmpty()) {
                        // Town on its own line. Left intact: the app's behaviour here is
                        // unverified, so a multi-word own-line town must stay a real assertion.
                        town = lines.remove(lines.size() - 1);
                    }
                }
            }

            String destination = lines.isEmpty() ? EMPTY : lines.get(0);
            return new Address(name, destination, town, postcode);
        }

        /** "Bishop Auckland" -> "Bishop". Used only to mirror the app's town truncation. */
        private static String firstToken(String value) {
            String[] tokens = value.trim().split("\\s+");
            return tokens.length == 0 ? EMPTY : tokens[0];
        }
    }
}
